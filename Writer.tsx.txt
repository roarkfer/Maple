import { useEffect, useState } from "react";
import { BookMarked, ChevronDown, ChevronLeft, ChevronUp, Download, Plus, Save, Tag, Trash2 } from "lucide-react";
import type { Notebook, ReaderSettings } from "@/lib/library-types";

type Props = {
  notebook: Notebook;
  settings: ReaderSettings;
  onChange: (text: string) => void;
  onTagsChange?: (tags: string[]) => void;
  onBack: () => void;
};

export default function Writer({ notebook, settings, onChange, onTagsChange, onBack }: Props) {
  const [text, setText] = useState(notebook.text);
  const [saved, setSaved] = useState(false);
  const [showTags, setShowTags] = useState(false);
  const [newTag, setNewTag] = useState<string | null>(null);

  const tags = notebook.tags ?? [];

  useEffect(() => setText(notebook.text), [notebook.id]);

  const save = () => {
    onChange(text);
    setSaved(true);
    setTimeout(() => setSaved(false), 1500);
  };

  const commitTag = () => {
    const t = (newTag ?? "").trim();
    if (t && !tags.includes(t)) onTagsChange?.([...tags, t]);
    setNewTag(null);
  };

  const openExportPage = () => {
    onChange(text);
    try {
      sessionStorage.setItem(
        "maple-notebook-export",
        JSON.stringify({ name: notebook.name, text }),
      );
      window.location.assign("/export");
    } catch {
      // La navegación permite volver a intentarlo incluso si el almacenamiento está restringido.
      window.location.assign("/export");
    }
  };

  return (
    <div className="fixed inset-0 z-50 flex flex-col bg-background text-foreground">
      <div className="flex items-center gap-2 px-2 py-2">
        <button
          type="button"
          onClick={() => {
            onChange(text);
            onBack();
          }}
          aria-label="Volver"
          className="flex size-10 shrink-0 items-center justify-center rounded-lg border border-border text-muted-foreground active:scale-95"
        >
          <ChevronLeft size={26} strokeWidth={2} />
        </button>
        <span className="flex-1 truncate text-[13px] text-muted-foreground">{notebook.name}</span>
        <button
          type="button"
          onClick={openExportPage}
          aria-label="Descargar"
          className="flex items-center gap-1 rounded-md border border-border px-2 py-2 text-[13px] active:scale-95"
        >
          <Download size={15} strokeWidth={1.8} />
        </button>
        <button
          type="button"
          onClick={save}
          aria-label="Guardar"
          className="flex items-center gap-1 rounded-md border border-foreground px-2 py-2 text-[13px] active:scale-95"
        >
          <Save size={15} strokeWidth={1.8} /> {saved ? "Guardado" : "Guardar"}
        </button>
        <button
          type="button"
          onClick={() => setShowTags((v) => !v)}
          aria-expanded={showTags}
          className="flex items-center gap-1 rounded-md border border-border px-2 py-2 text-[13px] active:scale-95"
        >
          <BookMarked size={15} strokeWidth={1.8} /> Grimorio
          {showTags ? <ChevronUp size={13} /> : <ChevronDown size={13} />}
        </button>
      </div>

      {showTags && (
        <div className="mx-2 mb-2 rounded-md border border-border p-2">
          <ul className="space-y-1">
            {tags.map((t) => (
              <li key={t} className="flex items-center gap-2 text-[15px]">
                <Tag size={14} strokeWidth={1.8} className="text-muted-foreground" />
                <span className="flex-1 truncate">{t}</span>
                <button
                  type="button"
                  aria-label={`Quitar ${t}`}
                  onClick={() => onTagsChange?.(tags.filter((x) => x !== t))}
                  className="text-muted-foreground"
                >
                  <Trash2 size={14} strokeWidth={1.8} />
                </button>
              </li>
            ))}
            {tags.length === 0 && newTag === null && (
              <li className="text-[14px] text-muted-foreground">Sin etiquetas.</li>
            )}
            {newTag !== null && (
              <li className="flex items-center gap-2">
                <Tag size={14} strokeWidth={1.8} className="text-muted-foreground" />
                <input
                  autoFocus
                  value={newTag}
                  onChange={(e) => setNewTag(e.target.value)}
                  onBlur={commitTag}
                  onKeyDown={(e) => {
                    if (e.key === "Enter") {
                      e.preventDefault();
                      commitTag();
                    }
                  }}
                  placeholder="Nueva etiqueta…"
                  className="flex-1 bg-transparent py-1 text-[15px] outline-none placeholder:text-muted-foreground"
                />
              </li>
            )}
          </ul>
          <button
            type="button"
            aria-label="Agregar etiqueta"
            onClick={() => setNewTag("")}
            className="mt-2 flex size-8 items-center justify-center rounded-md border border-foreground active:scale-95"
          >
            <Plus size={16} strokeWidth={1.9} />
          </button>
        </div>
      )}

      <textarea
        value={text}
        onChange={(e) => setText(e.target.value)}
        placeholder="Escribe…"
        autoFocus
        style={{ fontSize: settings.fontSize, lineHeight: 1.6 }}
        className="w-full flex-1 resize-none bg-transparent px-4 pb-6 outline-none placeholder:text-muted-foreground"
      />
    </div>
  );
}
